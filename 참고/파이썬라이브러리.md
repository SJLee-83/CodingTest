# map
map은 리스트의 모든 원소에 함수를 하나씩 적용해주는 함수

map(적용할_함수, 반복가능한_것)

nums = ['1', '2', '3']
result = map(int, nums)      # 각 원소에 int()를 적용
print(list(result))          # [1, 2, 3]
